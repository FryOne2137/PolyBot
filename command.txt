# 1. Dłuższy warm-up strukturalnego belief world od zera. Smoke checkpoint
# miał tylko 1000 pozycji; nie zamrażamy po nim losowego jeszcze backbone'u.
# Wykresy: runs\belief\train (kolejne uruchomienie: train2, train3, ...).
python scripts/train_belief.py `
  --model-out models\belief-structured-v1.pt `
  --positions 500000 `
  --batch-size 64 `
  --learning-rate 2e-4 `
  --min-hidden-tiles 48 `
  --max-moves-per-world 128 `
  --map-size 11 `
  --map-type lakes `
  --random-seed `
  --device cuda

# 2. Bootstrap policy/value. Uczy m.in. ruchu w stronę widocznej wioski,
# którego gęsty natywny packet akcji SelfPlayPool nie koduje jako osobnej cechy.
# Wykresy: runs\teacher\train.
python scripts/train_teacher.py `
  --model-in models\belief-structured-v1.pt `
  --model-out models\teacher-structured-v1.pt `
  --games 300 `
  --updates-per-game 8 `
  --batch-size 64 `
  --learning-rate 5e-5 `
  --map-size 11 `
  --map-type lakes `
  --random-seed `
  --device cuda

# 3. Główny trening: pełny ISMCTS w SelfPlayPool.
# Jeśli zabraknie VRAM, zmniejsz tylko --num-envs z 128 do 64.
# Wykresy: runs\self_play_pool\train.
python scripts/train_self_play_pool.py `

  --anchor-model models\teacher-structured-v1.pt `
  --anchor-policy-weight 0.20 `
  --model-out models\selfplay-ismcts-v1.pt `
  --num-envs 128 `
  --steps 10000 `
  --updates 5000 `
  --num-threads 16 `
  --max-pending-leaves 4 `
  --visible-action-history 0 `
  --belief-retries 4 `
  --simulations 32 `
  --particles 2 `
  --mcts-max-actions 1024 `
  --strategic-prior-weight 1.0 `
  --teacher-prior-weight 0.20 `
  --batch-size 64 `
  --replay-capacity 100000 `
  --belief-weight 0.25 `
  --bridge-constraint-weight 1.0 `
  --bridge-replay-fraction 0.25 `

# 5. Liga populacyjna: wszystkie trzy fazy (self-play kandydata, gry PFSP
# przeciw populacji i sparowane bramki) działają w szybkim SelfPlayPool.
# Modele są routowane przez agent_id bez wolnego GameEnv/MctsPool. Checkpointy
# po promocji trafią
# do models\league\, pełna historia do models\league\league.sqlite3, a wykresy
# do runs\league\train\round_001, round_002 itd.
# Stały --seed jest bazą wielu map, nie jednej mapy. Każda runda ma odległy
# seed self-play, a bramka osobny, nieużywany w treningu zestaw map. Wznów
# następny run od ostatniego champion-XXX.pt.
python scripts/train_league.py `
  --champion-in models\teacher-structured-v1.pt `
  --checkpoint-dir models\league `
  --checkpoint-prefix champion `
  --league-db models\league\league.sqlite3 `

  --population-games 64 `
  --population-updates-per-game 2 `
  --population-workers 16 `
  --pfsp-alpha 2.0 `
  --champion-probability 0.40 `
  --minimum-sampling-probability 0.05 `
  --rounds 3 `
  --promotion-games 100 `
  --promotion-score 0.55 `
  --promotion-lower-bound 0.50 `
  --historical-games 20 `
  --population-score 0.52 `
  --historical-score-floor 0.45 `
  --elo-k 32 `
  --evaluation-seed 1500000007 `
  --evaluation-seed-stride 20000033 `
  --round-seed-stride 10000019 `
  --num-envs 128 `
  --steps 1000 `
  --updates 500 `
  --num-threads 16 `
  --max-pending-leaves 4 `
  --visible-action-history 0 `
  --belief-retries 4 `
  --simulations 32 `
  --particles 2 `
  --mcts-max-actions 1024 `
  --batch-size 64 `
  --replay-capacity 50000 `
  --max-replay-epochs 6 `
  --belief-weight 0.25 `
  --bridge-constraint-weight 1.0 `
  --bridge-replay-fraction 0.25 `
  --exploration-rate 0.10 `
  --teacher-policy-target-weight 0.10 `
  --td-lambda 1.0 `
  --learning-rate 1e-5 `
