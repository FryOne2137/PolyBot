# 1. Dłuższy warm-up strukturalnego belief world.
# Kontynuuje policy/value z checkpointu, a dodatkowo uczy dekodery mapy.
python scripts/train_belief.py `
  --model-in models\belief-structured-smoke.pt `
  --model-out models\belief-structured-v1.pt `
  --positions 500000 `
  --batch-size 64 `
  --learning-rate 2e-4 `
  --min-hidden-tiles 48 `
  --max-moves-per-world 128 `
  --map-size 11 `
  --map-type lakes `
  --random-seed `
  --device cuda

# 2. Główny trening: pełny ISMCTS w SelfPlayPool.
# Jeśli zabraknie VRAM, zmniejsz tylko --num-envs z 128 do 64.
python scripts/train_self_play_pool.py `
  --model-in models\belief-structured-v1.pt `
  --model-out models\selfplay-ismcts-v1.pt `
  --num-envs 128 `
  --steps 10000 `
  --updates 5000 `
  --num-threads 16 `
  --max-pending-leaves 4 `
  --visible-action-history 64 `
  --belief-retries 4 `
  --simulations 32 `
  --particles 2 `
  --mcts-max-actions 1024 `
  --batch-size 64 `
  --learning-rate 3e-5 `
  --map-size 11 `
  --map-type lakes `
  --random-seed `
  --device cuda

# 3. Uczciwa ewaluacja kandydata przeciw checkpointowi wejściowemu.
python scripts/evaluate.py `
  --model-a models\selfplay-ismcts-v1.pt `
  --model-b models\belief-structured-v1.pt `
  --games 100 `
  --simulations 32 `
  --particles 2 `
  --mcts-max-actions 1024 `
  --max-moves 400 `
  --device cuda

# 4. Liga: jeden kandydat na raz, stała bramka przeciw championowi.
# Zapisane zostaną wyłącznie checkpointy po promocji w models\league\.
# Każda runda ma nowy seed self-play, a wszystkie bramki ten sam zestaw
# seedów ewaluacyjnych. Wznów następny run od ostatniego champion-XXX.pt.
python scripts/train_league.py `
  --champion-in models\belief-structured-v1.pt `
  --checkpoint-dir models\league `
  --checkpoint-prefix champion `
  --rounds 10 `
  --promotion-games 100 `
  --promotion-score 0.55 `
  --evaluation-seed 91337 `
  --num-envs 128 `
  --steps 1000 `
  --updates 500 `
  --num-threads 16 `
  --max-pending-leaves 4 `
  --visible-action-history 64 `
  --belief-retries 4 `
  --simulations 32 `
  --particles 2 `
  --mcts-max-actions 1024 `
  --batch-size 64 `
  --learning-rate 3e-5 `
  --max-moves 400 `
  --map-size 11 `
  --map-type lakes `
  --random-seed `
  --device cuda
