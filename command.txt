# Windows PowerShell: concurrent MCTS actors with one shared CUDA inference queue.

  --model-out models\candidate-fast.pt `

  --updates-per-game 8 `
  --opponent mixed `
  --mixed-opponent heuristic `

  --teacher-replay-fraction 0.25 `
  --learning-rate 5e-5 `
  --exploration-rate 0.05 `

  --simulations 64 `
  --particles 4 `
  --inference-batch-size 32 `
  --mcts-max-actions 512 `
  --workers 16 `
  --mcts-threads 1 `
  --global-inference-batch-size 256 `
